  retry_command() {
    local attempts="$1"
    local delay_seconds="$2"
    shift 2

    local attempt=1
    until "$@"; do
      local status="$?"
      if [[ "$attempt" -ge "$attempts" ]]; then
        return "$status"
      fi
      echo "Command failed with status $status. Retrying in ${delay_seconds}s ($attempt/${attempts})..."
      sleep "$delay_seconds"
      attempt=$((attempt + 1))
    done
  }
